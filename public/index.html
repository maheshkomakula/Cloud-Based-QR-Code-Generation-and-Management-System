Node.js / JavaScript Fetch Example 
// Create a Dynamic QR Code via API
const response = await fetch('http://localhost:5000/api/qrcodes', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    title: 'Spring Product Launch',
    type: 'url',
    destinationUrl: 'https://myshop.com/new-product',
    isDynamic: true,
    customShortCode: 'spring26'
  })
});
const data = await response.json();
console.log('Short Link:', `http://localhost:5000/r/${data.data.shortCode}`);
